log() { printf '%s\n' "$*" >&2; }
die() { log "错误：$*"; exit 1; }
now_ms() { jq -nr 'now * 1000 | floor'; }

# 仅返回 JSON；网络、HTTP 或业务错误时立即退出，不重试 POST。
request_json() {
  local method="$1" url="$2" body="${3:-}" http_code error_message
  local args=(
    --silent --show-error --retry 0
    --connect-timeout "$CONNECT_TIMEOUT" --max-time "$REQUEST_TIMEOUT"
    --request "$method" --url "$url"
    --output "$RESPONSE_FILE" --write-out '%{http_code}'
    --header "accept: $HEADER_ACCEPT"
    --header "accept-language: $HEADER_ACCEPT_LANGUAGE"
    --header "authorization: $HEADER_AUTHORIZATION"
    --header "clienttype: $HEADER_CLIENTTYPE"
    --cookie "$HEADER_COOKIE"
    --header "deviceid: $HEADER_DEVICEID"
    --header "priority: $HEADER_PRIORITY"
    --header "referer: $HEADER_REFERER"
    --header "sec-ch-ua: $HEADER_SEC_CH_UA"
    --header "sec-ch-ua-mobile: $HEADER_SEC_CH_UA_MOBILE"
    --header "sec-ch-ua-platform: $HEADER_SEC_CH_UA_PLATFORM"
    --header "sec-fetch-dest: $HEADER_SEC_FETCH_DEST"
    --header "sec-fetch-mode: $HEADER_SEC_FETCH_MODE"
    --header "sec-fetch-site: $HEADER_SEC_FETCH_SITE"
    --header "user-agent: $HEADER_USER_AGENT"
  )
  if [[ "$method" == 'POST' ]]; then
    args+=(--header "content-type: $HEADER_CONTENT_TYPE"
           --header "origin: $HEADER_ORIGIN" --data-raw "$body")
  fi
  if ! http_code=$(curl "${args[@]}"); then
    die "$method 网络请求未确认成功，已停止；请核对状态后再运行。"
  fi
  [[ "$http_code" =~ ^2[0-9][0-9]$ ]] || die "$method 返回 HTTP $http_code，已停止。"
  if ! jq -e 'type == "object"' "$RESPONSE_FILE" >/dev/null 2>&1; then
    die "$method 响应不是 JSON 对象，已停止。"
  fi
  if ! jq -e '(.code == 0 or .code == "0") and (.success != false)' "$RESPONSE_FILE" >/dev/null; then
    error_message=$(jq -r '.message // .msg // ("code=" + (.code | tostring))' "$RESPONSE_FILE")
    die "$method 接口失败：$error_message"
  fi
  cat "$RESPONSE_FILE"
}

fetch_list() {
  local stamp course_id_encoded result
  stamp=$(now_ms)
  course_id_encoded=$(jq -nr --arg value "$COURSE_ID" '$value | @uri')
  result=$(request_json GET "$BASE_URL/course/courseClient/coursewareList?courseId=$course_id_encoded&ver=$VER&timestamp=$stamp") || return 1
  if ! jq -e '
    .data | (type == "array") and
    all(.[]; type == "object" and (.id | type == "string" and length > 0) and has("isLearned")) and
    ([.[].id] | length == (unique | length))
  ' <<< "$result" >/dev/null; then
    die '课程列表结构异常：预期 data 为数组，每项具有唯一 id 和 isLearned。'
  fi
  printf '%s\n' "$result"
}

learned_state() {
  local result="$1" courseware_id="$2"
  jq -er --arg id "$courseware_id" '
    [.data[] | select(.id == $id)] |
    if length != 1 then error("课程列表中找不到唯一课件：" + $id)
    else (.[0].isLearned | tostring | ascii_downcase) end |
    if . == "true" or . == "1" then "done"
    elif . == "false" or . == "0" then "pending"
    else error("无法识别 isLearned，停止上报") end
  ' <<< "$result"
}

make_payload() {
  local courseware_id="$1" report_time="$2" stamp="$3"
  jq -cn \
    --arg courseWareId "$courseware_id" --arg courseId "$COURSE_ID" \
    --argjson time "$report_time" --argjson progress "$PROGRESS" --argjson status "$STATUS" \
    --arg bizType "$BIZ_TYPE" --arg bizId "$BIZ_ID" --arg isExchange "$IS_EXCHANGE" \
    --argjson timeStamp "$stamp" \
    '[{courseWareId: $courseWareId, courseId: $courseId, time: $time,
       progress: $progress, status: $status, bizType: $bizType, bizId: $bizId,
       isExchange: $isExchange, timeStamp: $timeStamp}]'
}

main() {
  local tool snapshot initial_rows initial_row total index=0
  local courseware_id name state report_time attempt stamp payload
  for tool in curl jq mktemp; do
    command -v "$tool" >/dev/null 2>&1 || die "缺少依赖 $tool，请先安装。"
  done
  [[ -n "$TOKEN" ]] || die '请填写文件开头的 TOKEN，或设置 EDU_TOKEN。'
  [[ -n "$HEADER_DEVICEID" ]] || die '请填写 HEADER_DEVICEID。'
  [[ "$INITIAL_TIME" =~ ^[1-9][0-9]*$ && "$MAX_TIME" =~ ^[1-9][0-9]*$ ]] || die 'INITIAL_TIME 和 MAX_TIME 必须为正整数。'
  (( MAX_TIME >= INITIAL_TIME )) || die 'MAX_TIME 不能小于 INITIAL_TIME。'
  [[ "$CHECK_INTERVAL" =~ ^[0-9]+([.][0-9]+)?$ ]] || die 'CHECK_INTERVAL 必须为非负秒数。'
  [[ "$PROGRESS" =~ ^[0-9]+$ && "$STATUS" =~ ^[0-9]+$ ]] || die 'PROGRESS 和 STATUS 必须为非负整数。'

  RESPONSE_FILE=$(mktemp)
  trap 'rm -f -- "$RESPONSE_FILE"' EXIT
  trap 'exit 130' INT
  trap 'exit 143' TERM

  snapshot=$(fetch_list)
  total=$(jq -r '.data | length' <<< "$snapshot")
  if [[ "$total" == 0 ]]; then log '课程列表为空。'; return 0; fi
  initial_rows=$(jq -c '.data[]' <<< "$snapshot")
  log "共 $total 个课件；每个未完成课件从 $INITIAL_TIME 开始，封顶 $MAX_TIME。"

  while IFS= read -r initial_row; do
    index=$((index + 1))
    courseware_id=$(jq -r '.id' <<< "$initial_row")
    name=$(jq -r '(.cwName // .id) | gsub("[\\r\\n]"; " ")' <<< "$initial_row")
    state=$(learned_state "$snapshot" "$courseware_id")
    if [[ "$state" == 'done' ]]; then
      log "[$index/$total] 已完成，跳过：$name"
      continue
    fi

    report_time=$INITIAL_TIME
    attempt=1
    while [[ "$state" != 'done' ]]; do
      stamp=$(now_ms)
      payload=$(make_payload "$courseware_id" "$report_time" "$stamp")
      log "[$index/$total] $name | 第 $attempt 次上报，time=$report_time"
      request_json POST "$BASE_URL/course/viewDuration/reportLearnTime?ver=$VER&timestamp=$stamp" "$payload" >/dev/null

      sleep "$CHECK_INTERVAL"
      snapshot=$(fetch_list)
      state=$(learned_state "$snapshot" "$courseware_id")
      if [[ "$state" == 'done' ]]; then
        log "[$index/$total] 已完成：$name，开始下一个课件。"
        break
      fi

      if (( report_time < MAX_TIME )); then
        report_time=$((report_time * 2))
        if (( report_time > MAX_TIME )); then report_time=$MAX_TIME; fi
      fi
      attempt=$((attempt + 1))
    done
  done <<< "$initial_rows"
  log '所有课件均已完成。'
}

if [[ "${BASH_SOURCE[0]}" == "$0" ]]; then
  main "$@"
fi
