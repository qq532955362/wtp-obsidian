### 刷新nacos配置文件

`需要前置命令jq`

```sh
yum install jq
```

`/root/data-migrate/nacos-refresh-config.sh`

```sh
#!/bin/bash

# --- UTF-8 安全的 urlencode（依赖 jq >= 1.6）---
urlencode() {
    local input="$1"
    [ -z "$input" ] && return
    printf '%s' "$input" | jq -sRr '@uri'
}

# ================== 配置区 ==================
NACOS_SERVER="http://172.20.242.120:8848"                 # Nacos 地址
NAMESPACE_ID="9abf7d79-7a53-4ee4-9154-af29137b9639"       # 命名空间ID（public 留空 ""）
GROUP="DEFAULT_GROUP"                                     # 分组

USERNAME="nacos"                                          # 如启用认证，填用户名（如 nacos）
PASSWORD="Myatoto_nacos.kd8qc"                            # 密码（如 nacos）
AUTH_HEADER=""

# 示例 dataId 列表
DATA_IDS=(
'atoto-product-core-prod.yml'
'atoto-gateway-admin-prod.yml'
'atoto-product-search-prod.yml'
'atoto-product-ibook-prod.yml'
'atoto-common-file-prod.yml'
'atoto-user-prod.yml'
'atoto-software-upgrade-prod.yml'
'atoto-mall-index-prod.yml'
'atoto-mall-product-prod.yml'
'atoto-erp-core-prod.yml'
'atoto-erp-flowable-prod.yml'
'atoto-file-center-prod.yml'
'atoto-knowledge-chat-prod.yml'
'atoto-gps-core-prod.yml'
'atoto-pricing-system-prod.yml'
'atoto-data-crawler-prod.yml'
'atoto-data-sync-prod.yml'
'atoto-point-service-prod.yml'
)

# =============== 认证（如需要）===============
if [ -n "$USERNAME" ] && [ -n "$PASSWORD" ]; then
  echo "🔑 正在认证..."
  LOGIN_RESP=$(curl -s -X POST "${NACOS_SERVER}/nacos/v1/auth/users/login" \
    -d "username=${USERNAME}&password=${PASSWORD}")
  TOKEN=$(echo "$LOGIN_RESP" | jq -r '.accessToken // empty')
  if [ -z "$TOKEN" ]; then
    echo "❌ 认证失败"
    exit 1
  fi
  AUTH_HEADER="-H Authorization: Bearer $TOKEN"
fi

# =============== 主循环 =====================
for dataId in "${DATA_IDS[@]}"; do
  echo "→ 刷新配置: $dataId"

  # 构造 GET URL（注意：用 tenant 传 namespaceId）
  get_url="${NACOS_SERVER}/nacos/v1/cs/configs?show=all&dataId=$(urlencode "$dataId")&group=$(urlencode "$GROUP")"
  [ -n "$NAMESPACE_ID" ] && get_url="${get_url}&tenant=$(urlencode "$NAMESPACE_ID")"

  # 获取完整 JSON 响应
  full_response=$(curl -s $AUTH_HEADER "$get_url")

  # 检查是否返回错误（如 404）
  if echo "$full_response" | jq empty >/dev/null 2>&1; then
    # 提取 content 字段（原始配置内容）
    content=$(echo "$full_response" | jq -r '.content // empty')
	echo "当前data_id:${dataId}的配置文件内容是：${content}"
    if [ -z "$content" ]; then
      echo "  ⚠️ 配置内容为空"
      continue
    fi
  else
    echo "  ❌ 获取失败: $full_response"
    continue
  fi

  # 准备 POST 数据（必须 urlencode content！）
  post_data="type=yaml&dataId=$(urlencode "$dataId")&group=$(urlencode "$GROUP")&content=$(urlencode "$content")"
  [ -n "$NAMESPACE_ID" ] && post_data="${post_data}&tenant=$(urlencode "$NAMESPACE_ID")"

  # 重新发布
  resp=$(curl -s $AUTH_HEADER \
    -X POST "${NACOS_SERVER}/nacos/v1/cs/configs" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    --data "$post_data")

  if [ "$resp" = "true" ]; then
    echo "  ✅ 刷新成功"
  else
    echo "  ❌ 刷新失败: $resp"
  fi

  sleep 0.1
done

echo "🎉 完成！"

```
