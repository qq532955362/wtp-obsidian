#### 1.nacos的config_info表配置修改
操作如下
```sql
-- 1 复制一份config_info表到config_info_data_migrate
CREATE TABLE config_info_data_migrate LIKE config_info;

INSERT INTO config_info_data_migrate
SELECT * FROM config_info;

-- 设置变量
set @source_str = 'jdbc:mysql://172.20.242.122:3306';
-- 测试的链接 创建了目标库之后记得修改
set @target_str = 'jdbc:mysql://172.20.242.122:3333';
set @target_ns = 'e5a232b8-b406-4750-a67f-b4e0ff446a89';
set @spring_profile = 'test';

-- 查询连接了数据库的服务的服务名
SELECT DISTINCT REPLACE(data_id, CONCAT('-', @spring_profile, '.yml'), '')
FROM
  config_info_data_migrate
WHERE
  tenant_id = @target_ns
  AND content LIKE CONCAT('%', @source_str, '%');
```
